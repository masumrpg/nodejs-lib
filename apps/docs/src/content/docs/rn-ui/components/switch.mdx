export const exampleCode = `import { useState } from 'react';
import { Switch, Box, Text } from '@masumdev/rn-ui';

export function SwitchExample() {
  const [enabled, setEnabled] = useState(true);

  return (
    <Box row align="center" justify="space-between">
      <Text>Enable Notifications</Text>
      <Switch
        value={enabled}
        onValueChange={setEnabled}
        tone="primary"
        size="md"
        haptic
      />
    </Box>
  );
}`;

`Switch` toggles the state of a single setting on or off with smooth UI thread worklet translation, custom tone color tracks, round and oval variants, and haptic taps.
