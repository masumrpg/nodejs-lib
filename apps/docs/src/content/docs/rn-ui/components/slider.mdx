export const sliderCode = `import React, { useState } from 'react';
import { Slider, Box, Text } from '@masumdev/rn-ui';

export function SliderExample() {
  const [volume, setVolume] = useState(40);

  return (
    <Box gap="md">
      <Text>Volume: {Math.round(volume)}%</Text>
      <Slider
        min={0}
        max={100}
        step={1}
        value={volume}
        onValueChange={setVolume}
        tone="primary"
        haptic
      />
    </Box>
  );
}`;

`Slider` provides smooth 60–120fps continuous or stepped numeric input along a horizontal track with Reanimated worklets, gesture detection, accessibility adjustable role, and haptic detents.
