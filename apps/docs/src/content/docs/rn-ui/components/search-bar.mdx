export const exampleCode = `import { useState } from 'react';
import { SearchBar } from '@masumdev/rn-ui';

export function SearchBarExample() {
  const [query, setQuery] = useState('');

  return (
    <SearchBar
      placeholder="Search documents or members..."
      value={query}
      onChangeText={setQuery}
      onSearchDebounced={(cleanQuery) => console.log('Searching for:', cleanQuery)}
      debounceMs={300}
      showCancelButton
    />
  );
}`;

export const pillExample = `import { SearchBar } from '@masumdev/rn-ui';

<SearchBar
  placeholder="Filter contacts..."
  variant="outline"
  shape="pill"
  tone="accent"
  cancelLabel="Done"
/>`;

`SearchBar` provides an integrated search input featuring built-in debounced callbacks, an animated clear button, loading spinner states, and a smooth iOS/Material style cancel button transition.
