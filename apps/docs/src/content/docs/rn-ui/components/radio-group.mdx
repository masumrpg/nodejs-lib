export const exampleCode = `import { useState } from 'react';
import { RadioGroup, RadioGroupItem, Box } from '@masumdev/rn-ui';

export function RadioExample() {
  const [value, setValue] = useState('expo');

  return (
    <RadioGroup value={value} onValueChange={setValue} variant="card">
      <RadioGroupItem value="expo" label="Expo Framework" description="Universal React Native ecosystem" />
      <RadioGroupItem value="cli" label="React Native CLI" description="Standard bare native development" />
    </RadioGroup>
  );
}`;

`RadioGroup` allows users to select a single option from a set with accessible radiogroup roles, card and plain layouts, and fluid selection indicator animations.

## Usage

### RadioGroup
