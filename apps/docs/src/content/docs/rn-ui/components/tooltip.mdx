export const exampleCode = `import { Tooltip, IconButton } from '@masumdev/rn-ui';
import { Download } from 'lucide-react-native';

export function TooltipExample() {
  return (
    <Tooltip content="Download monthly balance statement" position="top">
      <IconButton
        icon={({ color, size }) => <Download color={color} size={size} />}
        variant="outline"
        tone="primary"
      />
    </Tooltip>
  );
}`;

export const pressExample = `import { Tooltip, Button } from '@masumdev/rn-ui';
import { Info } from 'lucide-react-native';

<Tooltip
  content="API key configured for production"
  triggerMode="press"
  position="top"
  tone="dark"
  duration={3000}
>
  <Button tone="secondary" variant="outline">
    View Key Status
  </Button>
</Tooltip>`;

`Tooltip` provides an anchored contextual balloon anchored to any trigger element, rendering a directional pointer arrow and handling touch-outside dismissal.
