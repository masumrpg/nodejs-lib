export const exampleCode = `import { BubbleGroup, Bubble, BubbleContent } from '@masumdev/rn-ui';

export function ChatExample() {
  return (
    <BubbleGroup>
      <Bubble align="start" tone="default" variant="soft">
        <BubbleContent>Hello! How can I help you today?</BubbleContent>
      </Bubble>
      <Bubble align="end" tone="primary" variant="filled">
        <BubbleContent>I'd like to check the RN UI docs.</BubbleContent>
      </Bubble>
    </BubbleGroup>
  );
}`;

`Bubble` renders chat message bubbles with orthogonal `tone` and `variant` styling for messaging interfaces.
