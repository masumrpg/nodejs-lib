export const exampleCode = `import { useState } from 'react';
import { Rating, Box, Text } from '@masumdev/rn-ui';

export function RatingExample() {
  const [rating, setRating] = useState(4.5);

  return (
    <Box gap="md">
      <Rating
        value={rating}
        onValueChange={setRating}
        max={5}
        precision="half"
        shape="star"
        tone="warning"
        showValue
        haptic
      />
      <Text color="textMuted">Current score: {rating} / 5</Text>
    </Box>
  );
}`;

`Rating` displays star, heart, thumb, fire, smile, or shield ratings with fluid tap and drag input powered by Gesture Handler 3 (`useTapGesture`, `usePanGesture`, `useCompetingGestures`), fractional precision (`full`, `half`, `exact`), and haptic snaps.
