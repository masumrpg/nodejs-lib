export const exampleCode = `import React from 'react';
import { Accordion, Box } from '@masumdev/rn-ui';

export function AccordionExample() {
  const items = [
    {
      id: '1',
      title: 'Is @masumdev/rn-ui open source?',
      content: 'Yes, it is 100% open source under the MIT license.',
    },
    {
      id: '2',
      title: 'Does it support dark mode?',
      content: 'Yes, dark mode is supported out of the box with ThemeProvider.',
    },
  ];

  return (
    <Box gap="lg">
      {/* Outline Variant */}
      <Accordion items={items} variant="outline" />

      {/* Glassmorphism Surface */}
      <Accordion items={items} variant="outline" glass />

      {/* Flat Borderless Variant */}
      <Accordion items={items} variant="flat" />
    </Box>
  );
}`;

`Accordion` renders vertical stacks of interactive collapsible panels ideal for FAQs, product specifications, and settings menus.

## Props API

<TypeTable
  type={{
    items: { type: 'AccordionItem[]', description: 'Array of accordion panel objects containing id, title, and content.' },
    variant: { type: "'outline' | 'flat' | 'ghost'", default: "'outline'", description: 'Visual container style variant.' },
    glass: { type: 'boolean', default: 'false', description: 'Enables translucent glass surface background styling.' },
    allowMultiple: { type: 'boolean', default: 'false', description: 'Determines whether multiple panels can remain expanded simultaneously.' },
    openIds: { type: 'string[]', description: 'Controlled array of open panel IDs.' },
    defaultOpenIds: { type: 'string[]', default: '[]', description: 'Uncontrolled initial open panel IDs.' },
    onOpenChange: { type: '(openIds: string[]) => void', description: 'Callback triggered when open panel selection changes.' },
  }}
/>
